CASE status
    WHEN 'planned'::text THEN ((dispatched_at IS NULL) AND (started_at IS NULL) AND (completed_at IS NULL) AND (cancelled_at IS NULL))
    WHEN 'ready'::text THEN ((dispatched_at IS NOT NULL) AND (started_at IS NULL) AND (completed_at IS NULL) AND (cancelled_at IS NULL))
    WHEN 'active'::text THEN ((dispatched_at IS NOT NULL) AND (started_at IS NOT NULL) AND (completed_at IS NULL) AND (cancelled_at IS NULL))
    WHEN 'completed'::text THEN ((dispatched_at IS NOT NULL) AND (started_at IS NOT NULL) AND (completed_at IS NOT NULL) AND (cancelled_at IS NULL))
    WHEN 'cancelled'::text THEN ((cancelled_at IS NOT NULL) AND (completed_at IS NULL) AND ((started_at IS NULL) OR (dispatched_at IS NOT NULL)))
    ELSE false
END)
